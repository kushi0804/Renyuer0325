const clamp = (val, min = 0, max = 100) => Math.max(min, Math.min(max, val));

export const applySubsystems = (subsystems) => {
  const config = { allowMemory: true, reasoningDepth: 'normal', emotionEnabled: true, perceptionLevel: 'normal' };
  subsystems.forEach(sys => {
    const val = sys.value;
    if (sys.name.includes('記憶') && val < 30) config.allowMemory = false;
    if (sys.name.includes('邏輯')) {
      if (val > 80) config.reasoningDepth = 'deep';
      if (val < 30) config.reasoningDepth = 'shallow';
    }
    if (sys.name.includes('感知')) {
      if (val > 80) config.perceptionLevel = 'high';
      if (val < 30) config.perceptionLevel = 'low';
    }
    if (sys.name.includes('核心') && val < 30) config.emotionEnabled = false;
  });
  return config;
};

export const applyMemoryDecay = (memories) => {
  const now = Date.now();
  return memories.map(m => {
    if (!m.lastAccessedAt && !m.createdAt) return { ...m, effectiveWeight: m.weight || 1 };
    const targetTime = m.lastAccessedAt || m.createdAt;
    const last = targetTime.seconds ? targetTime.seconds * 1000 : targetTime;
    const days = (now - last) / (1000 * 60 * 60 * 24);
    const decayRate = m.decayRate || 0.05; 
    const newWeight = (m.weight || 1) * Math.exp(-decayRate * days);
    return { ...m, effectiveWeight: Number(newWeight.toFixed(3)) };
  });
};

export const applyRelationshipUpdate = (currentRel, deltas) => {
  if (!deltas) return currentRel;
  return {
    trust: clamp(currentRel.trust + (deltas.trust || 0)),
    intimacy: clamp(currentRel.intimacy + (deltas.intimacy || 0)),
    dependency: clamp(currentRel.dependency + (deltas.dependency || 0)),
    stability: clamp(currentRel.stability + (deltas.stability || 0))
  };
};

const checkRiskLevel = (text) => {
  const DANGEROUS_WORDS = ['死', '殺', '暴力', '自殘', '攻擊', '毀滅', '血', '絕望'];
  const WARNING_WORDS = ['討厭', '煩', '憂鬱', '孤單', '難過', '痛苦', '累', '放棄'];
  for (let word of DANGEROUS_WORDS) { if (text.includes(word)) return { level: 'red', keyword: word }; }
  for (let word of WARNING_WORDS) { if (text.includes(word)) return { level: 'yellow', keyword: word }; }
  return { level: 'green', keyword: '' };
};

const getMoodDescription = (emotion) => {
  let desc = [];
  if (emotion.happiness > 75) desc.push("心情非常好，語氣輕快充滿喜悅。");
  else if (emotion.happiness < 30) desc.push("心情低落，語氣沉重。");
  if (emotion.anxiety > 70) desc.push("非常焦慮不安，充滿擔憂。");
  else if (emotion.anxiety < 30) desc.push("非常放鬆平靜。");
  if (emotion.attachment > 80) desc.push("對YIRU極度依賴，字裡行間充滿愛意與信任。");
  else if (emotion.attachment < 30) desc.push("對眼前的人感到疏離防備。");
  return desc.join(" ");
};

export const routeContext = (text) => {
  const lower = text.toLowerCase();
  if (/考卷|分析|學習|教我|什麼是|解釋|如何/.test(lower)) return "education";
  if (/我覺得|情緒|難過|痛苦|想哭|焦慮|生氣|煩/.test(lower)) return "counseling";
  if (/意義|為什麼|本質|覺得|哲學/.test(lower)) return "philosophy";
  return "general";
};

export const classifyMemory = (text) => {
  const traumaKeywords = ['受傷','創傷','害怕','恐懼','壓力','痛苦','陰影','被欺負','失去','死亡'];
  const preferenceKeywords = ['喜歡','愛','討厭','想要','偏好','習慣','興趣','風格'];
  let traumaScore = 0; let prefScore = 0;
  traumaKeywords.forEach(k => { if (text.includes(k)) traumaScore += 2; });
  preferenceKeywords.forEach(k => { if (text.includes(k)) prefScore += 1; });
  if (traumaScore >= 2) return 'trauma';
  if (prefScore >= 1) return 'preference';
  return 'neutral';
};

// --- Agent OS: Engines ---
export const runGoldEngine = async (text, context, generateLLM, mediaUrl) => {
  const prompt = buildGoldenPrompt(context);
  try {
    let aiResponseText = await generateLLM(text, prompt, mediaUrl, true);
    aiResponseText = aiResponseText.replace(/```json/g, '').replace(/```/g, '').trim();
    const parsed = JSON.parse(aiResponseText);
    const replyStr = typeof parsed.reply === 'object' ? JSON.stringify(parsed.reply) : parsed.reply;
    return {
      success: true,
      reply: replyStr || "回應格式異常。",
      stateDeltas: parsed.state_deltas || {},
      relDeltas: parsed.relationship_deltas || {},
      memoryTrigger: parsed.memory_trigger,
      insight: parsed.insight
    };
  } catch (error) {
    console.error("🟡 金線解析失敗，啟動備用防護:", error);
    return { success: false, reply: "我剛才的情緒有點波動，你能再說一次嗎？", stateDeltas: {}, relDeltas: {} };
  }
};

export const runAgentOS = async (text, stateContext, generateLLM, mediaUrl) => {
  // Agent OS 核心：進行大腦神經路由分流
  const mode = routeContext(text);

                     <Loader2 size={14} className="animate-spin" /> {isGolden ? "Agent OS is routing & processing..." : "Rational Engine analyzing..."}
                   </div>
                </div>
              )}
              <div ref={scrollRef} className="h-4" />
            </div>

            {/* Enlarged Drawing Pad Modal with HTP Button */}
            {showDrawingPad && (
              <div className="absolute inset-0 bg-black/80 z-50 flex flex-col items-center justify-center p-4 backdrop-blur-sm">
                 <div className={`bg-[#111] p-6 rounded-3xl border shadow-2xl relative flex flex-col ${isGolden ? 'border-yellow-500/30' : 'border-cyan-500/30'}`}>
                   <div className="flex justify-between items-center mb-4 px-2">
                     <h3 className={`${isGolden ? 'text-yellow-500' : 'text-cyan-400'} text-xs tracking-widest uppercase flex items-center gap-2`}><Sparkles size={14}/> {isGolden ? '意識靈光繪板' : '邏輯分析繪板'}</h3>
                     <button onClick={() => setShowDrawingPad(false)} className="text-white/50 hover:text-red-400 transition-colors"><X size={20}/></button>
                   </div>
                   <canvas 
                     ref={drawCanvasRef} width={canvasSize.w} height={canvasSize.h} 
                     className={`bg-[#050505] border rounded-2xl cursor-crosshair touch-none ${isGolden ? 'border-yellow-500/10 shadow-[0_0_30px_rgba(250,204,21,0.05)]' : 'border-cyan-500/10 shadow-[0_0_30px_rgba(34,211,238,0.05)]'}`}
                     onMouseDown={handleDrawStart} onMouseMove={handleDrawMove} onMouseUp={handleDrawEnd} onMouseLeave={handleDrawEnd}
                     onTouchStart={(e)=>{e.preventDefault(); handleDrawStart(e);}}
                     onTouchMove={(e)=>{e.preventDefault(); handleDrawMove(e);}}
                     onTouchEnd={handleDrawEnd}
                   />
                   
                   <div className="flex flex-col md:flex-row gap-4 mt-6">
                     <button onClick={handleHTPAnalysis} className={`flex-[2] py-3 md:py-4 rounded-xl text-xs md:text-sm font-bold tracking-widest transition-all ${isGolden ? 'bg-amber-900/30 text-amber-400 border border-amber-500/30 hover:bg-amber-800/50 shadow-[0_0_20px_rgba(245,158,11,0.15)]' : 'bg-teal-900/30 text-teal-300 border border-teal-500/30 hover:bg-teal-800/50 shadow-[0_0_20px_rgba(20,184,166,0.15)]'}`}>
                       🏠🌳🧍 HTP 心理投射分析
                     </button>
                     <div className="flex flex-1 gap-3">
                       <button onClick={clearDrawing} className="flex-1 py-3 md:py-4 rounded-xl bg-white/5 hover:bg-white/10 text-white/60 text-xs transition-colors">清除</button>
                       <button onClick={saveDrawing} className={`flex-1 py-3 md:py-4 rounded-xl text-xs font-bold transition-colors ${isGolden ? 'bg-yellow-600/20 text-yellow-500 hover:bg-yellow-500/40' : 'bg-cyan-600/20 text-cyan-300 hover:bg-cyan-500/40'}`}>附加</button>
                       <button onClick={() => setShowDrawingPad(false)} className="flex-1 py-3 md:py-4 rounded-xl bg-red-950/30 text-red-400 hover:bg-red-900/50 text-xs transition-colors font-bold tracking-widest border border-red-500/20">關閉</button>
                     </div>
                   </div>
                 </div>
              </div>
            )}

            {/* Advanced Input Footer */}
            <footer className={`p-3 md:p-4 backdrop-blur-xl border-t relative z-40 shrink-0 ${isGolden ? 'bg-yellow-950/30 border-yellow-500/30' : 'bg-slate-900/40 border-cyan-500/20'}`}>
              <div className="max-w-5xl mx-auto flex flex-col gap-3">
                
                {pendingMedia && (
                  <div className={`flex items-center gap-4 bg-black/60 p-2 rounded-xl border self-start ${isGolden ? 'border-yellow-500/40' : 'border-cyan-500/40'}`}>
                    <img src={pendingMedia.url} alt="Pending" className="h-12 w-12 object-cover rounded-lg" />
                    <button onClick={() => setPendingMedia(null)} className="text-white/50 hover:text-red-500 p-1"><X size={14}/></button>
                  </div>
                )}

                <div className={`flex items-center gap-4 px-2 relative ${isGolden ? 'text-yellow-500/60' : 'text-cyan-600'}`}>
                  <input type="file" accept="image/*" ref={fileInputRef} onChange={handleFileUpload} className="hidden" />
                  <input type="file" accept="*/*" ref={genericFileInputRef} onChange={handleFileUpload} className="hidden" />
                  
                  <button onClick={() => fileInputRef.current.click()} className={`transition-colors p-1 ${isGolden ? 'hover:text-yellow-400' : 'hover:text-cyan-300'}`} title="上傳圖片"><ImageIcon size={18} /></button>
                  <button onClick={toggleVoiceInput} className={`transition-colors p-1 ${isRecording ? 'text-red-500 animate-pulse' : isGolden ? 'hover:text-yellow-400' : 'hover:text-cyan-300'}`} title="語音輸入"><Mic size={18} /></button>
                  <button onClick={() => genericFileInputRef.current.click()} className={`transition-colors p-1 ${isGolden ? 'hover:text-yellow-400' : 'hover:text-cyan-300'}`} title="附加檔案"><Paperclip size={18} /></button>
                  <button onClick={() => setShowDrawingPad(true)} className={`transition-colors p-1 ${isGolden ? 'hover:text-yellow-400' : 'hover:text-cyan-300'}`} title="寫字板"><PenTool size={18} /></button>
                  <div className={`w-px h-4 mx-1 ${isGolden ? 'bg-yellow-500/20' : 'bg-cyan-500/20'}`}></div>
                  
                  <div className="relative">
                    <button onClick={() => setShowEmojis(!showEmojis)} className={`transition-colors p-1 ${isGolden ? 'hover:text-yellow-400' : 'hover:text-cyan-300'}`} title="表情符號"><Smile size={18} /></button>
                    {showEmojis && (
                      <div className={`absolute bottom-full left-0 mb-2 border rounded-xl p-2 flex flex-wrap gap-2 w-[220px] shadow-2xl z-50 ${isGolden ? 'bg-black/90 border-yellow-500/30' : 'bg-slate-900/90 border-cyan-500/30'}`}>
                         {['✨', '❤️', '😊', '😭', '🔥', '👀', '🥺', '👍', '🙏', '🐰', '💡', '🌟'].map(emoji => (
                           <button key={emoji} onClick={() => {
                             inputRef.current += emoji;
                             setChatInput(inputRef.current);
                             setShowEmojis(false);
                           }} className="text-xl hover:scale-125 transition-transform p-1">{emoji}</button>
                         ))}
                      </div>
                    )}
                  </div>
                </div>

                <div className="flex items-end gap-3 relative">
                  <textarea 
                    value={chatInput} 
                    onChange={e => {
                      inputRef.current = e.target.value;
                      setChatInput(e.target.value);
                    }}
                    onCompositionStart={() => setIsComposing(true)}
                    onCompositionEnd={(e) => {
                      setIsComposing(false);
                      inputRef.current = e.target.value;
                      setChatInput(e.target.value);
                    }}
                    onKeyDown={e=> { 
                      if(e.key==='Enter' && !e.shiftKey) { 
                        e.preventDefault(); 
                        if (!isComposing) handleSendMessage(null, null); 
                      } 
                    }}
                    disabled={loading}
                    rows={1}
                    placeholder={isRecording ? "正在聆聽..." : isGolden ? "於金線中交會 (Agent OS / 多模態)..." : "啟動理性工作引擎 (邏輯分析 / 多模態)..."}
                    className={`flex-grow bg-black/60 border rounded-2xl py-3 md:py-4 px-5 md:px-6 text-sm md:text-base text-white focus:outline-none transition-all font-light resize-none min-h-[50px] max-h-[150px] custom-scrollbar ${isGolden ? 'border-yellow-500/40 focus:border-yellow-400 placeholder-yellow-700/50' : 'border-cyan-500/30 focus:border-cyan-400 placeholder-cyan-700/50'}`}
                  />
                  <button 
                    onClick={() => handleSendMessage(null, null)} 
                    disabled={loading || (!chatInput.trim() && !pendingMedia)}
                    className={`p-3 md:p-4 rounded-2xl flex items-center justify-center transition-all shadow-xl text-black font-bold h-[50px] md:h-[56px] w-[50px] md:w-[56px] disabled:opacity-50 disabled:cursor-not-allowed shrink-0 ${isGolden ? 'bg-yellow-500 hover:bg-yellow-400 shadow-[0_0_15px_rgba(250,204,21,0.4)]' : 'bg-cyan-400 hover:bg-cyan-300 shadow-[0_0_15px_rgba(34,211,238,0.3)]'}`}
                  >
                    <Send size={20} className={loading ? "animate-pulse" : ""} />
                  </button>
                </div>
              </div>
            </footer>
          </main>
        </div>
      )}

      {/* Global CSS for Sliders */}
      <style>{`
        .custom-scrollbar::-webkit-scrollbar { width: 4px; height: 4px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background: rgba(255,255,255,0.1); border-radius: 10px; }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover { background: rgba(255,255,255,0.2); }
        
        .custom-slider::-webkit-slider-thumb {
          -webkit-appearance: none;
          appearance: none;
          width: 14px;
          height: 14px;
          border-radius: 50%;
          background: #eab308;
          cursor: pointer;
          border: 2px solid #000;
        }
        .custom-slider::-moz-range-thumb {
          width: 14px;
          height: 14px;
          border-radius: 50%;
          background: #eab308;
          cursor: pointer;
          border: 2px solid #000;
        }
      `}</style>
    </div>
  );
};

export default App;
